        for body in self.bodies:
            body.v += body.a * 0.5 * dt

function computeGradients() {
    // Reset acceleration vectors
    BODIES.forEach(b => { b.ax = 0; b.ay = 0; });

    for (let i = 0; i < BODIES.length; i++) {
        for (let j = 0; j < BODIES.length; j++) {
            if (i === j) continue;

            let bi = BODIES[i];
            let bj = BODIES[j];

            let dx = bj.x - bi.x;
            let dy = bj.y - bi.y;
            let distSq = dx*dx + dy*dy;
            let dist = Math.sqrt(distSq);

            // SDT FORMULA: a = Beta / r^2
            // Direction is normalized vector (dx/dist, dy/dist)
            let accel = bj.beta / distSq;

            bi.ax += (dx / dist) * accel;
            bi.ay += (dy / dist) * accel;
        }
    }
}

function stepPhysics() {
    // Velocity Verlet Integrator
    
    // 1. Half-step Velocity
    BODIES.forEach(b => {
        b.vx += b.ax * 0.5 * dt;
        b.vy += b.ay * 0.5 * dt;
    });

    // 2. Full-step Position
    BODIES.forEach(b => {
        b.x += b.vx * dt;
        b.y += b.vy * dt;
        
        // Update Trail
        if (time % 10 === 0) { // Save trail every 10 ticks
            b.trail.push({x: b.x, y: b.y});
            if (b.trail.length > 300) b.trail.shift();
        }
    });

    // 3. Recalculate Pressure Field (Acceleration)
    computeGradients();

    // 4. Half-step Velocity
    BODIES.forEach(b => {
        b.vx += b.ax * 0.5 * dt;
        b.vy += b.ay * 0.5 * dt;
    });

    time += dt;
}
